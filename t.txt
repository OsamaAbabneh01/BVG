public async Task<BVGResponse<object>> TakeSnapshot(SnapshotDto dto)
{
    var options = new TransactionOptions
    {
        IsolationLevel = System.Transactions.IsolationLevel.ReadCommitted,
        Timeout = TimeSpan.FromMinutes(30)
    };

    try
    {
        using var scope = new TransactionScope(
            TransactionScopeOption.Required,
            options,
            TransactionScopeAsyncFlowOption.Enabled);

        var year = int.Parse(dto.Year);
        var month = int.Parse(dto.Month);

        var cycle = await _cycleInfoRepository
            .GetFirstOrDefaultAsync(c =>
                c.CycleDate.Year == year &&
                c.CycleDate.Month == month);

        if (cycle == null)
        {
            cycle = new CycleInfo
            {
                CycleDate = new DateOnly(year, month, 1),
                FingerprintOk = "",
                ReferenceOk = "",
                ReferenceReason = "",
                Status = 0,
                StartDate = DateTime.UtcNow,
                EndDate = null,
                CreatedAt = DateTime.UtcNow,
                UpdatedAt = null
            };

            await _cycleInfoRepository.AddCycleInfo(cycle);
        }
        else
        {
            var isChecked = cycle.SnapshotBulks.Any(b => b.IsChecked);

            if (!isChecked)
            {
                return BVGResponse<object>.Fail(
                    BVGStatusCode.Conflict,
                    BVGErrorCode.VerificationRequired);
            }
        }

        var bulk = new SnapshotBulk
        {
            CycleId = cycle.Id,
            CreatedAt = DateTime.UtcNow,
            UpdatedAt = null,
            IsChecked = false
        };

        await _snapshotBulkRepository.AddSnapshotBulk(bulk);

        // Save changes in the first SQL Server instance
        await _unitOfWorkService.SaveChangesAsync();

        // Execute procedure in the second SQL Server instance
        await _snapshotRepository.TakeSnapshotAsync(
            cycle.Id,
            bulk.Id);

        // Mark the distributed transaction as successful
        scope.Complete();

        return BVGResponse<object>.Success(
            null,
            BVGMessage.ExecuteSuccessfully,
            BVGStatusCode.Ok);
    }
    catch (Exception ex)
    {
        _logger.LogError(ex, "TakeSnapshot failed");
        throw;
    }
}
